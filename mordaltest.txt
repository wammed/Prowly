def run_terminal_stream_mode():
    """1行ずつディレイを設けて標準出力へ流し、リアルタイムマスクの即時検知をテスト"""
    print("\n--- [ターミナル ストリーミング テスト開始] ---")
    print("※ リアルタイムマスク機能が有効な場合、即座に ***MASKED*** へ置換されるか確認してください。\n")
    time.sleep(1)

    lines = generate_log_lines()
    for line in lines:
        sys.stdout.write(line + "\n")
        sys.stdout.flush()
        time.sleep(0.8)

    print("\n--- [ストリーミング 終了] ---\n")

def run_file_generation_mode():
    """エディタ（VS Code / Neovim等）で開くための平文ファイルを生成"""
    filename = "secret_masker_test.log"
    print(f"\n--- [ファイル生成モード] ---")
    
    lines = generate_log_lines()
    full_content = "\n".join(lines) + "\n"
    
    with open(filename, "w", encoding="utf-8") as f:
        f.write(full_content)

    abs_path = os.path.abspath(filename)
    print(f"ファイルを作成しました: {abs_path}")
    print("\n検証手順:")
    print(f"  1. お使いのエディタで `{filename}` を開く")
    print("  2. SecretMaskerプラグイン/機能が平文のキーをインラインまたはハイライトでマスクするか確認")
    print(f"  3. テスト後はファイルを削除: rm {filename}\n")
